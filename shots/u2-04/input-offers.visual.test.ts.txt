/**
 * The input card's saved-login offers in a real browser over fixed fake data, light and dark. Set
 * `FY_INPUT_OFFERS_SHOTS_DIR` to keep a screenshot of each state.
 */
import { afterAll, beforeAll, describe, expect, it } from 'bun:test';
import { mkdir, mkdtemp, readFile, rm } from 'node:fs/promises';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import type { Browser } from 'playwright-core';
import { sceneBuilder, sharedChromium } from '@ferretry/app/testing/browser';

const packageRoot = import.meta.dir;
const shotsDir = process.env.FY_INPUT_OFFERS_SHOTS_DIR ?? '';
const modes = ['light', 'dark'] as const;
const states = ['offers', 'other', 'settled'] as const;

let browser: Browser;
let directory = '';
let server: ReturnType<typeof Bun.serve> | undefined;

const build = async (): Promise<{ readonly css: string; readonly js: string }> => {
  directory = await mkdtemp(join(tmpdir(), 'ferretry-input-offers-'));
  const child = Bun.spawn(
    [process.execPath, sceneBuilder, '--out', directory, '--entry', 'fixtures/input-offers-scene.tsx'],
    { cwd: packageRoot, stderr: 'pipe', stdout: 'pipe' },
  );
  const [stdout, stderr, code] = await Promise.all([
    new Response(child.stdout).text(),
    new Response(child.stderr).text(),
    child.exited,
  ]);
  if (code !== 0) throw new Error(`the scene did not build (${code}):\n${stderr}\n${stdout}`);
  const manifest = JSON.parse(stdout.trim().split('\n').at(-1) as string) as { css: string; js: string };
  return { css: await readFile(manifest.css, 'utf8'), js: await readFile(manifest.js, 'utf8') };
};

describe('the input card with saved-login offers', () => {
  beforeAll(async () => {
    browser = await sharedChromium();
    const { css, js } = await build();
    const page = `<!doctype html><html><head><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><style>${css}</style></head><body class="bg-bg text-fg"><div id="root"></div><script type="module" src="/app.js"></script></body></html>`;
    server = Bun.serve({
      hostname: '127.0.0.1',
      port: 0,
      fetch: request => {
        const path = new URL(request.url).pathname;
        if (path === '/app.js') return new Response(js, { headers: { 'content-type': 'text/javascript' } });
        if (path === '/') return new Response(page, { headers: { 'content-type': 'text/html; charset=utf-8' } });
        return new Response('not here', { status: 404 });
      },
    });
    if (shotsDir !== '') await mkdir(shotsDir, { recursive: true });
  }, 180_000);

  afterAll(async () => {
    server?.stop(true);
    if (directory !== '') await rm(directory, { force: true, recursive: true });
  });

  for (const mode of modes)
    for (const state of states)
      it(`draws the ${state} state in ${mode} without sideways scroll`, async () => {
        const origin = server?.url.origin ?? '';
        const context = await browser.newContext({ viewport: { width: 420, height: 720 }, reducedMotion: 'reduce' });
        try {
          await context.route('**/*', async route =>
            new URL(route.request().url()).origin !== origin ? await route.abort() : await route.continue(),
          );
          const tab = await context.newPage();
          await tab.goto(`${origin}/?theme=studio-${mode}&state=${state}`);
          await tab.waitForSelector('html[data-scene-ready="1"]');
          const metrics = await tab.evaluate<{ inner: number; scroll: number }>(
            '({ inner: innerWidth, scroll: document.documentElement.scrollWidth })',
          );
          expect(metrics.scroll).toBeLessThanOrEqual(metrics.inner);
          if (shotsDir !== '') await tab.screenshot({ path: join(shotsDir, `input-offers-${state}-${mode}.png`) });
        } finally {
          await context.close();
        }
      }, 120_000);
});
