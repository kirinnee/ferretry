  #!/bin/sh
  # ferretry-managed — regenerated when this machine's runtimes change; edits are overwritten
  # runtime codex-golden 00000000-0000-4000-8000-00000000000a (codex)

  if [ -z "${REFERENCED:-}" ]; then printf '%s\n' 'ferretry: REFERENCED is not set — this account takes it from Ferretry'\''s secret store (secret GOLDEN_KEY). Launch it through the daemon, or run: fy secret set GOLDEN_KEY' >&2; exit 1; fi

  export CODEX_HOME='/fy/fleet/homes/00000000-0000-4000-8000-00000000000a'
  export FIXED_LITERAL='a b'
  export FY_CREDENTIAL='codex-host'
  export FY_HOME='/fy'
  export FY_RUNTIME='codex-golden'

  # Codex's sign-in is the file Logins keeps: link this home's auth.json to it (a link already right is left alone).
  _fy_auth='/fy/fleet/homes/00000000-0000-4000-8000-00000000000a/auth.json'
  _fy_auth_target='/fy/state/logins/credentials/golden/auth.json'
  if [ -L "$_fy_auth" ] || [ ! -e "$_fy_auth" ]; then
    [ "$(readlink "$_fy_auth" 2>/dev/null)" = "$_fy_auth_target" ] || ln -sfn "$_fy_auth_target" "$_fy_auth" 2>/dev/null ||
      echo 'ferretry: could not link this runtime to its Codex sign-in; sign in when Codex asks' >&2
  fi

  # One live writer per conversation: claim the one these arguments resume before handing this pid to
  # the harness. Only a refusal (exit 75: another session is writing it) stops the launch; any other failure warns and launches anyway.
  _fy_claim=0
  for _fy_arg in "$@"; do
    case "$_fy_arg" in resume) _fy_claim=1 ;; esac
  done
  if [ "$_fy_claim" = 1 ]; then
    if command -v '/opt/ferretry/bin/fy' >/dev/null 2>&1; then
      '/opt/ferretry/bin/fy' history claim --harness codex --pid $$ --from-arguments -- "$@"
      _fy_status=$?
      if [ "$_fy_status" -eq 75 ]; then exit 75; fi
      if [ "$_fy_status" -ne 0 ]; then
        if [ -n "${FY_SESSION_ID:-}" ]; then
          echo "ferretry: could not check whether another session is writing this conversation (fy history claim exited $_fy_status), so not launching: this session must hold its conversation before it writes" >&2
          exit 75
        fi
        echo "ferretry: could not check whether another session is writing this conversation (fy history claim exited $_fy_status), so launching anyway" >&2
      fi
    else
      if [ -n "${FY_SESSION_ID:-}" ]; then
        echo "ferretry: fy was not found, so this session could not check whether another session is writing this conversation; not launching" >&2
        exit 75
      fi
      echo "ferretry: fy was not found, so this launch did not check whether another session is writing this conversation; launching anyway" >&2
    fi
  fi

  exec '/opt/harness/bin/codex' '--profile-flag' 'value with space' "$@"
