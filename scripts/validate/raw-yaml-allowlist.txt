# Product files that parse YAML with a reader of their own, each with why that is safe — and, under each, the exports
# that reach that parse and the only files that may load them.
#
# Enforced by scripts/validate/no-raw-yaml.sh. Every reader of text a person or an agent can write goes through the
# shared bounded reader (packages/std/src/yaml-text.ts, imported as @ferretry/std/yaml) or the value the kernel already
# checked; a file belongs here only when it IS that reader, or when what it parses can never be such a text in the
# daemon. Three kinds of line, exact both ways (a stale or missing line fails):
#
#   path  # reason                             the FILE may parse YAML itself
#   path::export  <- a.ts, b.ts  # reason      an EXPORT of that file reaches a parse; only those callers load it
#   path::export  <- *  # reason               open to every caller: the shared bounded reader alone
#
# A new importer of a listed parser is a use too (laundering): add it to the line only with why it is safe there.

packages/std/src/yaml-text.ts  # the shared bounded reader itself: size cap, alias budget by length, no YAML 1.1 tags, duplicate keys in one pass
packages/std/src/yaml-text.ts::readConfigText  <- *  # the sanctioned door: every read of a config text goes through it
packages/std/src/yaml-text.ts::readConfigTextToEdit  <- *  # the same bounded read, handing back the document an edit starts from
packages/std/src/yaml-text.ts::editConfigText  <- *  # bounded read, then a comment-preserving edit
packages/std/src/yaml-text.ts::setConfigKey  <- *  # bounded read, then one key set

packages/plugins/board/cli/lib/patch-document.ts  # CLI side: writes a patch document with the library's stringify and reads nothing with it; the text a person hands `fy card patch` is read through @ferretry/std/yaml

packages/plugins/conversation/app/render/fy-render-chart-model.ts  # app side: a chart fence in a message, parsed in the viewer's own browser; never a file the daemon reads
packages/plugins/conversation/app/render/fy-render-chart-model.ts::readChart  <- packages/plugins/conversation/app/render/fy-render-chart.tsx  # the chart renderer, in the viewer's own browser

packages/plugins/conversation/app/render/fy-render-map-model.ts  # app side: a map fence in a message, parsed in the viewer's own browser; never a file the daemon reads
packages/plugins/conversation/app/render/fy-render-map-model.ts::readMap  <- packages/plugins/conversation/app/render/fy-render-map.tsx  # the map renderer, in the viewer's own browser

packages/app/src/prose/highlight.ts  # highlight.js's YAML grammar: it colours code for display and never builds a value from the text
